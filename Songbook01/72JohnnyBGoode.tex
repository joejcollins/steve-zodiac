\begin{SBSong}{Johnny B. Goode}
  {Originally had `colored boy' in the lyrics, but Berry changed it to ensure radio play}
  {Chuck Berry}
  \SBChord{A}
  \SBChord{D}
  \SBChord{E7}
  \SBDirection{E7}
  \begin{SBVerse}
    De\ldots ep \Ch{A}{down} in Louisiana close to New Orleans
    
    Way back up in the woods among the evergreens
    
    There \Ch{D}{stood} a log cabin made of earth and wood
    
    Where \Ch{A}{lived} a country boy named Johnny B Goode
    
    Who \Ch{E7}{never} ever learned to read or write so well
    
    But he could \Ch{A}{play} the guitar just like ringin' a bell
  \end{SBVerse}
  \begin{SBChorus}
    \Ch{A}{Go!} Go! Go, Johnny go! Go! Go!
    
    Go, Johnny, Go! \Ch{D}{Go!} Go!
    
    Go, Johnny, Go! \Ch{A}{Go!} Go!
    
    Go, Johnny, Go! \Ch{E7}{Go!} Go! \Ch{D}{Johnny} B. \Ch{A}{Goode}
  \end{SBChorus}
  \begin{SBVerse}
    He used to \Ch{A}{carry} his guitar in a gunny sack
    
    Or sit beneath the tree by the railroad track
    
    Oh an \Ch{D}{engineer} could see him sitting in the shade
    
    \Ch{A}{Strummin'} to the rhythm that the drivers made
    
    \Ch{E7}{People} passing by they'd stop and say
    
    Oh \Ch{A}{my} but that little country boy can play
  \end{SBVerse}
  \begin{SBVerse}
    His \Ch{A}{mother} told him some day you will be a man
    
    And you will be the leader of a big old band
    
    \Ch{D}{Many} people coming from miles around
    
    And \Ch{A}{hear} you play your music till the sun goes down
    
    \Ch{E7}{Maybe} someday your name gonna be in lights
    
    Sayin' \Ch{A}{Johnny} be Goode tonight
  \end{SBVerse}
  \SBDirection{Chorus\\Chorus}
\end{SBSong}
