\begin{SBSong}{Sway}
  {$Id: 24Sway.tex 624 2011-07-10 12:20:41Z LudwigVanBoll@gmail.com $}
  {`Sway' is the English version of `?`Qui\'en ser\'a?', a 1953 mambo song}
  {Norman Gimbel}
  \SBChord{Dm}
  \SBChord{Gm}
  \SBChord{A}
  \SBChord{C}
  \SBChord{F}
  \SBDirection{Dm}
  \begin{SBVerse}
    When marimba rhythms \Ch{Gm}{start} to play

    \Ch{Gm}{Dance} with me, \Ch{Dm}{make} me sway
    
    \Ch{Dm}{Like} a lazy ocean \Ch{Gm}{hugs} the shore
    
    \Ch{A}{Hold} me close, \Ch{Dm}{sway} me more \Ch{stop}{}
  \end{SBVerse}
  \begin{SBVerse}
    Like a flower bending \Ch{Gm}{in} the breeze
    
    \Ch{Gm}{Bend} with me, \Ch{Dm}{sway} with ease
    
    \Ch{Dm}{When} we dance you have a \Ch{Gm}{way} with me
    
    \Ch{A}{Stay} with me, \Ch{Dm}{sway} with me \Ch{stop}{}
  \end{SBVerse}
  \begin{SBVerse}
    Other dancers may \Ch{C}{be} on the floor
    
    \Ch{C}{Dear} but my eyes will \Ch{F}{see} only you
    
    \Ch{F}{Only} you have that \Ch{A}{magic} technique
    
    \Ch{A}{When} we sway I go \Ch{Dm}{we\ldots ak} \Ch{stop}{}
  \end{SBVerse}
  \begin{SBVerse}
    I can hear the sounds of \Ch{Gm}{violins}
    
    \Ch{Gm}{Long} before \Ch{Dm}{it} begins
    
    \Ch{Dm}{Make} me thrill as only \Ch{Gm}{you} know how
    
    \Ch{A}{Sway} me smooth, \Ch{Dm}{sway} me now \Ch{stop~(not~last~time)}{}
  \end{SBVerse}
  \SBDirection{Repeat all}
  \SBEnd{\Ch{Gm}{You} know how
  
    \Ch{A}{Sway} me smooth, \Ch{Dm}{sway} me now

    \Ch{Gm}{You} know how
  
    \Ch{A}{Sway} me smooth, \Ch{Dm}{sway} me now cha-cha-cha \Ch{stop}{}}
\end{SBSong}
