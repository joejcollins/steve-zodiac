\begin{SBSong}{Waltzing Matilda}
  {Australian slang for traveling by foot with a `Matilda' (bag) slung over one's back}
  {Banjo Paterson and Christina Macpherson}
  \SBChord{C}
  \SBChord{G}
  \SBChord{Am}
  \SBChord{F}
  \SBChord{CB}
  \begin{SBVerse}
    \Ch{C}{Once} a jolly \Ch{G}{swagman} \Ch{Am}{camped} beside a \Ch{F}{billabong}

    \Ch{C}{Under} the shade of a \Ch{G}{coolibah} tree, 

    And he \Ch{C}{sang} as he \Ch{G}{watched} and \Ch{Am}{waited} 'till his \Ch{F}{billy} boiled, 

    \Ch{C}{``You'll} come a \Ch{Am}{waltzing},\Ch{G}{Matilda}, with \Ch{C}{me.''}
  \end{SBVerse}
  \begin{SBChorus}
    \Ch{C}{``Waltzing} Matilda, \Ch{F}{waltzing} Matilda, 

    \Ch{C}{You'll} \Ch{CB}{come} a \Ch{Am}{waltzing,} Ma\Ch{F}{til}da, with \Ch{G}{me.}'' 

    And he \Ch{C}{sang} as he \Ch{G}{watched} and \Ch{Am}{waited} 'till his \Ch{F}{billy} boiled, 

    \Ch{C}{``You'll} come a \Ch{Am}{waltzing}, \Ch{G}{Matilda}, with \Ch{C}{me}.'' 
  \end{SBChorus}
  \begin{SBVerse}
    \Ch{C}{Down} came a \Ch{G}{jumpbuck} to \Ch{Am}{drink} at the \Ch{F}{billabong}, 

    \Ch{C}{Up} jumped the swagman and \Ch{G}{grabbed} him with glee, 

    And he \Ch{C}{sang} as he \Ch{G}{stowed} that \Ch{Am}{jumpbuck} in his \Ch{F}{tucker} bag, 

    \Ch{C}{``You'll} come a \Ch{Am}{waltzing}, \Ch{G}{Matilda}, with \Ch{C}{me.}''
  \end{SBVerse} 
  \begin{SBVerse}
    \Ch{C}{Up} rode the \Ch{G}{squatter,} \Ch{Am}{mounted} on his \Ch{F}{thoroughbred,} 

    \Ch{C}{Down} came the troopers, \Ch{G}{one}, two, three. 

    ``Where's \Ch{C}{that} jolly \Ch{G}{jumpbuck} \Ch{Am}{you've} got in your \Ch{F}{tucker} bag? 

    \Ch{C}{You'll} come a \Ch{Am}{waltzing}, \Ch{G}{Matilda}, with \Ch{C}{me}.''
  \end{SBVerse} 
  \begin{SBVerse}
    \Ch{C}{Up} jumped the \Ch{G}{swagman,} \Ch{Am}{sprang} into the \Ch{F}{billabong}, 

    \Ch{C}{``You'll} never catch me \Ch{G}{alive,}'' said he. 

    And his \Ch{C}{ghost} may be \Ch{G}{heard} as you \Ch{Am}{pass} by that \Ch{F}{billabong}, 

    \Ch{C}``{You'll} come a \Ch{Am}{waltzing}, \Ch{G}{Matilda,} with \Ch{C}{me}.'' 
  \end{SBVerse} 
  \SBEnd{\SBDirection{Repeat chorus then the final line again - slowly}

    \Ch{C}{``You'll} come a \Ch{Am}{waltzing}, \Ch{G}{Matilda}, with \Ch{C}{me.}''}
\end{SBSong}
