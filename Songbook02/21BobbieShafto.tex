\begin{SBSong}{Bobby Shafto}
  {Sung by the supporters of Robert Shafto, an eighteenth-century MP}
  {Traditional}
  \SBChord{D}
  \SBChord{A}
  \begin{SBVerse}
	\Ch{D}{Bobby} Shafto's gone to sea,

	\Ch{A}{Silver} buckles on his knee;

	\Ch{D}{He'll} come back and marry me,

	\Ch{A}{Bonny} Bobby \Ch{D}{Shafto!}

	\sout{\Ch{A}{Bonny} Bobby \Ch{D}{Shafto!}}
  \end{SBVerse}
  \begin{SBVerse}
	\Ch{D}{Bobby} Shafto's bright and fair,

	\Ch{A}{Panning} out his yellow hair;

	\Ch{D}{He}'s my love for evermore,

	\Ch{A}{Bonny} Bobby \Ch{D}{Shafto!}

	\sout{\Ch{A}{Bonny} Bobby \Ch{D}{Shafto!}}
  \end{SBVerse}
  \begin{SBVerse}
	\Ch{D}{Bobby} Shafto's getten a bairn,

	\Ch{D}{On} his arm and on his knee,
