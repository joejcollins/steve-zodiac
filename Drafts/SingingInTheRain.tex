  {Arthur Freed and Nacio Herb Brown}

  \SBChord{G}
  \SBIntro{Doo-dloo-doo-doo-doo
    
    Doo-dloo-doo-doo-doo-doo 
    
    Doo-dloo-doo-doo-doo-doo 
    
    Doo-dloo-doo-doo-doo-doo\dots}  
  \begin{SBVerse}
	\Ch{C}{I'm} singing in the rain
	
	Just singing in the rain
	
	What a glorious feeling
	
	I'm \Ch{G}{happy} again

	I'm laughing at clouds
	
	So dark up above
	
	the sun's in my heart
	
	And I'm \Ch{C}{ready} for love

	Let the stormy clouds chase
	
	Everyone from the place
	
	Come on with the rain
	
	I've a \Ch{G}{smile} on my face

	I'll walk down the lane
	
	With a happy refrain
	
	Just singing
	
	just singing in the rain

	'Cause I'm singing
	
	just singing in the \Ch{C}{rain}.
  \end{SBVerse}
