\begin{SBSong}{My Favorite Things}
  {$Id: FavoriteThings.tex 634 2011-07-12 09:53:53Z LudwigVanBoll@gmail.com $}
  {Julie Andrews favorite film is `The Americanization of Emily' (1964)}
  {Richard Rodgers and Oscar Hammerstein}
  \SBChord{Am}
  \SBChord{F}
  \SBChord{G}
  \SBChord{E7}
  \begin{SBVerse} 
	\Ch{Am}{Raindrops} on roses and whiskers on kittens;
	
	\Ch{F}{Bright} copper kettles and warm woolen mittens;
	
	\Ch{Am}{Brown} paper packages \Ch{G}{tied} up with strings;
	
	\Ch{F}{These} are a few of my \Ch{E7}{favorite} things.
  \end{SBVerse}
  \begin{SBVerse}
	\Ch{Am}{Cream}-colored ponies and crisp apple strudels;
	
	\Ch{F}{Doorbells} and sleigh bells and schnitzel with noodles;
	
	\Ch{Am}{Wild} geese that fly with the \Ch{G}{moon} on their wings;
	
	\Ch{F}{These} are a few of my \Ch{E7}{favorite} things.
  \end{SBVerse}
  \begin{SBVerse}
	\Ch{Am}{Girls} in white dresses with blue satin sashes;
	
	\Ch{F}{Snowflakes} that stay on my nose and eyelashes;
	
	\Ch{Am}{Silver}-white winters that \Ch{G}{melt} into springs;
	
	\Ch{F}{These} are a few of my \Ch{E7}{favorite} things.
  \end{SBVerse}
  \begin{SBVerse}
	\Ch{Am}{When} the dog bites,
	
	\Ch{G}{When} the bee stings,
	
	\Ch{F}{When} I'm feeling \Ch{Am}{sad},
	
	I simply remember my \Ch{G}{favorite} things,
	
	\Ch{F}{And} then I don't feel so \Ch{G}{bad}.
  \end{SBVerse}
\end{SBSong}
